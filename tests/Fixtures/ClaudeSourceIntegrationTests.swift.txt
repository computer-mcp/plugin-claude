private func claudeMCPResult(_ result: JSONValue) -> [String: JSONValue]? {
  result.objectValue?["structuredContent"]?.objectValue?["result"]?
    .objectValue?["structuredContent"]?.objectValue?["result"]?.objectValue
}

private func claudePayload(_ result: JSONValue) throws -> [String: JSONValue] {
  try #require(result.objectValue?["isError"] != .bool(true), "\(result)")
  var value = try #require(result.objectValue?["structuredContent"]?.objectValue?["result"])
  if let nested = value.objectValue?["structuredContent"]?.objectValue?["result"] {
    try #require(value.objectValue?["isError"] != .bool(true), "\(value)")
    value = nested
  }
  return try #require(value.objectValue)
}

private func claudePermissionDenied(_ result: JSONValue) throws {
  try #require(result.objectValue?["isError"] == .bool(true), "\(result)")
  let error = try #require(
    result.objectValue?["structuredContent"]?.objectValue?["error"]?.objectValue)
  try #require(
    [JSONValue.string("permission.required"), .string("permission.denied")].contains(
      error["code"] ?? .null), "\(result)")
}

private func claudeDenied(_ operation: @autoclosure () async throws -> JSONValue) async throws {
  let result = try await operation()
  try #require(
    !String(describing: result).contains("permission.catalog_required"),
    "A missing catalog is not denial coverage: \(result)")
  let outer = result.objectValue
  let nested = outer?["structuredContent"]?.objectValue?["result"]?.objectValue
  try #require(outer?["isError"] == .bool(true) || nested?["isError"] == .bool(true), "\(result)")
}

private func withClaudeHarness(
  _ package: PluginPackage, caller: String = "write", provider: String = "write", use: Bool = true,
  skills: Bool = true, operation: (ClaudeHarness) async throws -> Void
) async throws {
  let h = try await ClaudeHarness.make(
    package, caller: caller, provider: provider, use: use, skills: skills)
  do { try await operation(h) } catch {
    await h.diagnoseFailure()
    do { try await h.close() } catch {
      Issue.record("Claude fixture cleanup failed; retain root \(h.base.root): \(error)")
    }
    throw error
  }
  try await h.close()
}

  let deniedOperations: [String]
}

      }
      try fixtureApprovePlugin(
        runtimePackage, database: base.database, runtime: base.runtime.permissionRuntime,
        permissions: permissions(provider, use: false, skills: false))
      let enabledState = try await store.snapshot()
      let plugin = try PluginResolver.resolve(
        package: runtimePackage, source: origin,
        settings: try #require(enabledState.settings["claude"]),
        dependencyExecutables: [
          "claude-code": peer, "python-runtime": URL(fileURLWithPath: python),
        ], hostVersion: PluginVersion(ComputerMCPCLI.version), architecture: PluginHost.architecture
      )
      try #require(
        plugin.diagnostics.isEmpty && plugin.cliCommands.count == 1 && plugin.mcpServers.count == 1
          && plugin.skillRoots.count == 1)
      let riskConfiguration = GatewayConfiguration(mcp: .init(servers: plugin.mcpServers))
      let reviewed = try PluginContributionRequirements(
        sourceDigest: PluginStore.manifestDigest(package.manifest),
        declarations: .init(package.manifest), componentID: "stream")
      let server = try #require(plugin.mcpServers.first)
      let overrides = try #require(reviewed.contribution).toolRequirements
      try #require(Set(server.toolRisks.keys) == Set(overrides.keys))
      for name in overrides.keys {
        try #require(reviewed.requirements(for: name)?.project == .read)
        try #require(
          riskConfiguration.mcpRisk(for: .init(serverID: "claude-stream", toolName: name))
            == .readOnly)
      }
      for name in [
        "claude.run", "claude.run.start", "claude.run.cancel", "claude.run.release",
        "fixture.unlisted",
      ] {
        try #require(reviewed.requirements(for: name)?.project == .write)
        try #require(
          riskConfiguration.mcpRisk(for: .init(serverID: "claude-stream", toolName: name))
            == .externalWrite)
      }
      let consumerID = try #require(GatewayProfileID(rawValue: "claude-consumer"))
      let admin = try #require(
        base.database.profiles().first { $0.id == base.configuration.runtime.profileID })
      try #require(consumerID != .localAdmin && consumerID != admin.id)
      var callerPermissions = permissions(caller, use: use, skills: skills)
      // Explicit denial is representable in profile configuration and overrides initial Skills access.
      if !skills { try callerPermissions.set(.denied, for: .skills) }
      let consumer = ProfileGrant(
        id: consumerID, permissions: callerPermissions,
        allowedCallers: [.localMCP])
      try base.database.saveProfile(consumer)
      let gateway = try GatewayRuntime(
        configuration: .init(
          runtime: .init(caller: .localMCP, profileID: consumerID),
          profiles: base.configuration.profiles + [.init(grant: consumer)],
          skills: .init(enabled: true)),
        context: .init(
          caller: .localMCP, profileID: consumerID, projectID: projects[0],
          trustedPrincipalID: "claude-consumer"), database: base.database,
        registeredProjects: base.database.projects(), platform: base.platform, plugins: [plugin],
        bundledPlugins: .init(packages: [], issues: []))
      pending = gateway
      _ = try gateway.makeControlSession()
      let storedConsumer = try #require(base.database.profiles().first { $0.id == consumerID })
      try #require(
        storedConsumer.permissions.state(for: .skills) == (skills ? .granted(.access) : .denied)
      )
      try #require(base.database.profiles().first { $0.id == admin.id } == admin)
      let owner = try GatewayPermissionOwner(
        context: .init(
          caller: .localApp, profileID: .localAdmin, trustedPrincipalID: "claude-fixture-owner"),
        validate: {})
      let approval = try #require(
        gateway.permissionRuntime.service.view(owner: owner).providers.first {
          $0.identity == .package("claude")
        })
      try #require(approval.enabled && approval.fullAccessConsent == nil)
      let harness = ClaudeHarness(
        base: base, lifecycle: lifecycle, gateway: gateway, owner: owner, consumerID: consumerID,
        projects: projects, paths: paths, outside: outside)
      if use { try await harness.requireCatalog() }
      try #require(harness.callCount == 0)
      return harness
    } catch {
      await pending?.shutdown()
      do {
        try await cleanupClaude(base, lifecycle: lifecycle, projects: projects, paths: paths)
      } catch { Issue.record("Claude setup cleanup failed; retain \(base.root): \(error)") }
      throw error
    }
  }

  var callCount: Int {
    (try? String(contentsOf: worktree.appendingPathComponent("native-calls.jsonl"), encoding: .utf8)
      .split(separator: "\n").count) ?? 0
  }

  func catalog() async throws -> JSONValue {
    try await gateway.callToolForMCPAsync(
      name: "mcp.tools.list", arguments: .object(["server": .string("claude-stream")]),
      context: .empty)
  }

  func requireCatalog() async throws {
    let response = try await catalog()
    try #require(response.objectValue?["isError"] != .bool(true), "\(response)")
    let tools = try #require(
      response.objectValue?["structuredContent"]?.objectValue?["result"]?.arrayValue)
    let names = tools.compactMap { $0.objectValue?["name"]?.stringValue }
    try #require(names.count == 7)
    for tool in tools {
      let name = try #require(tool.objectValue?["name"]?.stringValue)
      let inspection = ["claude.run.list", "claude.run.result", "claude.run.events"].contains(name)
      try #require(
        tool.objectValue?["annotations"]?.objectValue?["readOnlyHint"] == .bool(inspection))
      if inspection {
        try #require(
          tool.objectValue?["_meta"]?.objectValue?["io.github.computer-mcp/risk"]
            == .string("read-only"))
      }
    }
    try #require(
      Set(names) == [
        "claude.run", "claude.run.start", "claude.run.list", "claude.run.result",
        "claude.run.events", "claude.run.cancel", "claude.run.release",
      ])
  }

  func mcp(_ name: String, _ arguments: [String: JSONValue] = [:]) async throws -> JSONValue {
    let response = try await consumerRouter.callToolForMCPAsync(
      name: "mcp.tools.call",
      arguments: .object([
        "server": .string("claude-stream"), "tool": .string(name), "arguments": .object(arguments),
      ]), context: .empty)
    lastMCPResponse = response
    return response
  }

  func diagnoseFailure() async {
    func log(_ label: String, _ value: String) {
      print("Claude failure diagnostic \(label): \(String(value.prefix(16384)))")
    }
    if let lastMCPResponse { log("last MCP response", String(describing: lastMCPResponse)) }
    for name in ["native-calls.jsonl"] {
      do {
        let file = try FileHandle(forReadingFrom: worktree.appendingPathComponent(name))
        defer { try? file.close() }
        let data = try file.read(upToCount: 16384) ?? Data()
        log(name, String(decoding: data, as: UTF8.self))
      } catch { log(name, String(describing: error)) }
    }
    do {
      let status = try await gateway.callToolForMCPAsync(
        name: "mcp.servers.status", arguments: .object(["server": .string("claude-stream")]),
        context: .empty)
      log("provider status", String(describing: status))
    } catch { log("provider status error", String(describing: error)) }
  }

  func cli(_ prompt: String) async throws -> JSONValue {
    let tool = try #require(
      gateway.listTools(context: .empty).first {
        $0.meta?.objectValue?["cli"]?.objectValue?["registration"] == .string("claude-cli")
          && $0.meta?.objectValue?["cli"]?.objectValue?["command"] == .string("print")
      })
    return try await gateway.callToolForMCPAsync(
      name: tool.name,
      arguments: .object(["prompt": .string(prompt), "permission_mode": .string("dontAsk")]),
      context: .empty)
  }

  func skill() async throws -> JSONValue {
    try await gateway.callToolForMCPAsync(
      name: "skills.read",
      arguments: .object(["name": .string("claude-code"), "root_id": .string("claude-skills")]),
      context: .empty)
  }

  func waitReady(_ marker: String = "native-ready") async throws {
    for _ in 0..<500 {
      if FileManager.default.fileExists(atPath: worktree.appendingPathComponent(marker).path) {
        return
      }
      try await Task.sleep(for: .milliseconds(20))
    }
    throw ClaudeConsumerError.checkpoint
  }

  func waitJoined() async throws {
    for _ in 0..<500 {
      try await gateway.permissionRuntime.coordinator.reconcile()
      var live = false
      for ownership in base.observations.ownerships {
        if try await !ownership.members().isEmpty { live = true }
      }
      if !live { return }
      try await Task.sleep(for: .milliseconds(20))
    }
    throw ClaudeConsumerError.cleanup
  }

  func settled(_ id: String) async throws -> [String: JSONValue] {
    for _ in 0..<200 {
      let result = try await mcp("claude.run.result", ["run_id": .string(id)])
      let value = try #require(claudeMCPResult(result))
      if value["completed"] == .bool(true) { return value }
      try await Task.sleep(for: .milliseconds(20))
    }
    throw ClaudeConsumerError.checkpoint
  }

  func cancelRetained() async throws {
    let id = try #require(
      try claudePayload(await mcp("claude.run.start", ["prompt": .string("hold")]))["run_id"]?
        .stringValue)
    try await waitReady()
    try await claudeDenied(await mcp("claude.run.release", ["run_id": .string(id)]))
    _ = try claudePayload(await mcp("claude.run.cancel", ["run_id": .string(id)]))
    var settled: [String: JSONValue] = [:]
    for _ in 0..<500 {
      // A cancelled run is intentionally a downstream tool error with retained data.
      let response = try await mcp("claude.run.result", ["run_id": .string(id)])
      settled =
        response.objectValue?["structuredContent"]?.objectValue?["result"]?.objectValue?[
          "structuredContent"]?.objectValue?["result"]?.objectValue ?? [:]
      if settled["completed"] == .bool(true) { break }
      try await Task.sleep(for: .milliseconds(20))
    }
    try #require(
      settled["cleanup_confirmed"] == .bool(true) && settled["state"] == .string("cancelled"))
    let listing = try claudePayload(await mcp("claude.run.list"))
    try #require(listing["runs"]?.arrayValue?.count == 1)
    _ = try claudePayload(await mcp("claude.run.release", ["run_id": .string(id)]))
    try #require(
      try claudePayload(await mcp("claude.run.list"))["runs"]?.arrayValue?.isEmpty == true)
  }

  func removeSelected() async throws {
    _ = try claudePayload(
      await lifecycle.callToolForMCPAsync(
        name: "git.worktree_remove",
        arguments: .object(["project_id": .string(projects[0]), "force": .bool(true)]),
        context: .init(session: base.session)))
    removed.insert(projects[0])
  }

  func close() async throws {
    await gateway.shutdown()
    try await cleanupClaude(
      base, lifecycle: lifecycle, projects: projects.filter { !removed.contains($0) }, paths: paths)
  }
}

private enum ClaudeConsumerError: Error { case checkpoint, cleanup }

private func cleanupClaude(
  _ base: FixedHostFixture, lifecycle: GatewayStandaloneGenerations, projects: [String],
  paths: [URL]
) async throws {
  for ownership in base.observations.ownerships {
    try #require(try await ownership.members().isEmpty)
  }
  for project in projects.reversed() {
    _ = try claudePayload(
      await lifecycle.callToolForMCPAsync(
        name: "git.worktree_remove",
        arguments: .object(["project_id": .string(project), "force": .bool(true)]),
        context: .init(session: base.session)))
  }
  for path in paths { try #require(!FileManager.default.fileExists(atPath: path.path)) }
  try await base.requireNoScratch()
  await lifecycle.shutdown()
  base.session.end()
  await base.runtime.shutdown()
  for ownership in base.observations.ownerships {
    try #require(try await ownership.members().isEmpty)
  }
  try FileManager.default.removeItem(at: base.root)
  try #require(!FileManager.default.fileExists(atPath: base.root.path))
}
